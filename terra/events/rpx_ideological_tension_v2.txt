# Ideological Tensions with Neighbor - Faction Aware
# This event represents the rising tensions between a totalitarian country and its neighbors, potentially leading to demands for change or even war. The MTTH and AI decision are heavily influenced by the relative strength of the factions involved, as well as the current war situation.
# Should create limited wars against neighbours
country_event = {
	id = ideological_tension_v2.1
	title = ideological_tension_v2.1.t
	desc = ideological_tension_v2.1.d
	picture = GFX_report_event_fascist_gathering
	
	trigger = {
		OR = {
			has_government = fascism
			has_government = communism
		}
		any_neighbor_country = {
			NOT = { has_government = ROOT }
			NOT = { has_war_with = ROOT }
			NOT = { is_in_faction_with = ROOT }
			is_subject = no
		}
		
		# Only after April 1936
		date > 1938.4.1
	}
	
	mean_time_to_happen = {
		days = 365
		
		modifier = {
			factor = 0.8
			has_war = no
		}
		
		modifier = {
			factor = 0.5
			has_idea = partial_economic_mobilisation
		}
		
		modifier = {
			factor = 0.6
			has_idea = war_economy
		}
		
		# We are significantly stronger than our enemies
		modifier = {
			factor = 0.6
			has_war = yes
			surrender_progress < 0.1
			enemies_strength_ratio < 0.7
		}
		
		# We are struggling in current war
		modifier = {
			factor = 3.0
			has_war = yes
			enemies_strength_ratio > 1.2
		}
		
		# We are losing badly
		modifier = {
			factor = 10.0
			has_war = yes
			OR = {
				surrender_progress > 0.3
				enemies_strength_ratio > 2.0
			}
		}
		
		# Check if any neighbor's faction is much weaker
		modifier = {
			factor = 0.5
			any_neighbor_country = {
				NOT = { has_government = ROOT }
				NOT = { has_war_with = ROOT }
				NOT = { is_in_faction_with = ROOT }
				is_subject = no
				OR = {
					# Target not in faction and we are
					AND = {
						NOT = { is_in_faction = yes }
						ROOT = { is_in_faction = yes }
					}
					# Or they're in a small faction
					AND = {
						is_in_faction = yes
						faction_leader = {
							num_faction_members < 3
						}
					}
				}
			}
		}
		
		# Slower if we face strong neighbor factions
		modifier = {
			factor = 2.0
			any_neighbor_country = {
				NOT = { has_government = ROOT }
				NOT = { has_war_with = ROOT }
				NOT = { is_in_faction_with = ROOT }
				is_subject = no
				is_in_faction = yes
				faction_leader = {
					num_faction_members > 5
				}
			}
		}
	}
	
	immediate = {
		# IMPORTANT: Save the country receiving the event first
		ROOT = {
			save_event_target_as = event_receiver
		}
		
		# Find a neighbor to potentially target
		random_neighbor_country = {
			limit = {
				NOT = { has_government = ROOT }
				NOT = { has_war_with = ROOT }
				NOT = { is_in_faction_with = ROOT }
				is_subject = no
			}
			save_event_target_as = potential_target
			save_event_target_as = comparison_target
		}
		
		# Calculate faction strength ratio using scripted effect
		calculate_faction_strength_ratio = yes
		
		# Invert the ratio for decision modifiers
		if = {
			limit = { check_variable = { faction_strength_ratio > 0 } }
			set_variable = { mtth_factor = 1 }
			divide_variable = { mtth_factor = faction_strength_ratio }
		}
	}
	
	# Demand they change their ways
	option = {
		name = ideological_tension_v2.1.a
		ai_chance = {
			factor = 70
			
			# More likely if we're winning our current war
			modifier = {
				factor = 1.5
				has_war = yes
				enemies_strength_ratio < 0.6
			}
			
			# Less likely if struggling
			modifier = {
				factor = 0.3
				has_war = yes
				enemies_strength_ratio > 1.3
			}
			
			# Use the ratio directly - higher ratio = more aggressive
			modifier = {
				factor = faction_strength_ratio
				check_variable = { faction_strength_ratio > 0.5 }
				check_variable = { faction_strength_ratio < 2.5 }
			}
			
			# Clamp extreme values
			modifier = {
				factor = 2.5
				check_variable = { faction_strength_ratio > 2.5 }
			}
			
			modifier = {
				factor = 0.5
				check_variable = { faction_strength_ratio < 0.5 }
			}
			
			# Target is losing their war
			modifier = {
				factor = 1.5
				event_target:potential_target = {
					has_war = yes
					enemies_strength_ratio > 1.2
				}
			}
			
			# We are losing badly
			modifier = {
				factor = 0.0
				has_war = yes
				OR = {
					surrender_progress > 0.3
					enemies_strength_ratio > 2.0
				}
			}
		}
		
		# FIXED: Create wargoal against the SAVED target, not a new random one
		create_wargoal = {
			type = puppet_wargoal_focus
			target = event_target:potential_target
		}
		
		add_political_power = -50
	}
	
	# Ignore for now
	option = {
		name = ideological_tension_v2.1.b
		ai_chance = {
			factor = 30
			
			# Much more likely if we're struggling
			modifier = {
				factor = 5.0
				has_war = yes
				enemies_strength_ratio > 1.3
			}
			
			# Inverse of the aggressive option - weaker faction = more likely to decline
			modifier = {
				factor = mtth_factor
				check_variable = { mtth_factor > 0.4 }
				check_variable = { mtth_factor < 2.0 }
			}
			
			# Clamp extreme values
			modifier = {
				factor = 2.0
				check_variable = { mtth_factor > 2.0 }
			}
			
			modifier = {
				factor = 0.4
				check_variable = { mtth_factor < 0.4 }
			}
		}
		
		add_stability = 0.05
	}
}
