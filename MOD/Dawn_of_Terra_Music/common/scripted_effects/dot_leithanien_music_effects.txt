# Start the appropriate track only when the player's war state changes.
# Weighted Shuffle selects between both war recordings after the first one.
dot_leithanien_music_sync = {
    if = {
        limit = { tag = LTH is_ai = no }
        if = {
            limit = {
                has_war = yes
                NOT = { has_country_flag = dot_lth_music_war }
            }
            scoped_play_song = "dawn_leithanien_der_hexenkoenig"
            set_country_flag = dot_lth_music_war
            clr_country_flag = dot_lth_music_peace
        }
        else_if = {
            limit = {
                has_war = no
                NOT = { has_country_flag = dot_lth_music_peace }
            }
            scoped_play_song = "dawn_leithanien_visage"
            set_country_flag = dot_lth_music_peace
            clr_country_flag = dot_lth_music_war
        }
    }
}
