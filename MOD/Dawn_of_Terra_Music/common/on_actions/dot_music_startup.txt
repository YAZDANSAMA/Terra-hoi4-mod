on_actions = {
    on_startup = {
        effect = {
            every_country = {
                limit = { is_ai = no }
                dot_music_player_initialize = yes
            }
        }
    }
    # Explicit country scope avoids attacker/defender scope ambiguity.
    on_war = {
        effect = {
            VCT = { dot_victoria_music_sync = yes }
            URS = { dot_ursus_music_sync = yes }
            LTH = { dot_leithanien_music_sync = yes }
        }
    }
    on_war_relation_added = {
        effect = {
            VCT = { dot_victoria_music_sync = yes }
            URS = { dot_ursus_music_sync = yes }
            LTH = { dot_leithanien_music_sync = yes }
        }
    }
    on_peace = {
        effect = {
            VCT = { dot_victoria_music_sync = yes }
            URS = { dot_ursus_music_sync = yes }
            LTH = { dot_leithanien_music_sync = yes }
        }
    }
    # Country-scoped daily fallback handles delayed relation updates and old saves.
    on_daily = {
        effect = {
            # Initialize an older save once on its next daily pulse.
            if = {
                limit = { is_ai = no NOT = { has_country_flag = dot_music_004_initialized } }
                dot_music_player_initialize = yes
            }
            if = {
                limit = { tag = VCT is_ai = no }
                dot_victoria_music_sync = yes
            }
            if = {
                limit = { tag = URS is_ai = no }
                dot_ursus_music_sync = yes
            }
            if = {
                limit = { tag = LTH is_ai = no }
                dot_leithanien_music_sync = yes
            }
        }
    }
}
