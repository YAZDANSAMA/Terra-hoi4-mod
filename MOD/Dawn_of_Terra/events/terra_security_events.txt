add_namespace = terra_security

# This event is sent directly from the warning country to the threatened human.
# ROOT is the recipient and FROM is the sender. It does not declare a war.
country_event = {
    id = terra_security.1
    title = terra_security.1.t
    desc = terra_security.1.d
    picture = GFX_report_event_generic_read_write
    is_triggered_only = yes
    trigger = {
        is_ai = no
        terra_dip_ready = yes
        FROM = { terra_security_actor_allowed = yes }
        NOT = { has_war_with = FROM }
    }
    timeout_days = 30
    option = {
        name = terra_security.1.a
        custom_effect_tooltip = terra_security_warning_notice_tt
    }
    option = {
        name = terra_security.1.b
        trigger = { terra_dip_talks_available_from = yes has_political_power > 14 }
        if = {
            limit = { terra_dip_talks_available_from = yes has_political_power > 14 }
            add_political_power = -15
            terra_dip_change_relation_effect = { PARTNER = FROM TRUST = 5 GRIEVANCE = -5 }
            FROM = { terra_dip_change_relation_effect = { PARTNER = ROOT TRUST = 5 GRIEVANCE = -5 } }
        }
        custom_effect_tooltip = terra_security_warning_channel_tt
    }
}
