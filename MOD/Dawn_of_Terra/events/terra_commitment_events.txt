    id = terra_promises.106
    hidden = yes
    is_triggered_only = yes
    immediate = {
        if = {
            limit = {
                is_ai = yes
                terra_contract_open_from = yes
                has_political_power > 24
                has_equipment = { infantry_equipment > 999 }
                NOT = { terra_dip_rejects_from = yes }
                NOT = { terra_dip_has_betrayal_from = yes }
                NOT = { terra_dip_claim_conflict_from = yes }
                OR = { terra_dip_trusts_from = yes terra_dip_has_common_interest_from = yes }
            }
            if = {
                limit = {
                    has_country_flag = terra_dip_personality_guardian
                    NOT = { is_in_faction_with = FROM }
                    NOT = { has_guaranteed = FROM }
                    DIPLOMACY_GUARANTEE_ENABLE_TRIGGER = yes
                    num_divisions > 10
                    FROM = { num_divisions < 15 }
                }
                add_political_power = -25
                terra_contract_offer_defense_effect = yes
            }
            else = {
                add_political_power = -25
                terra_contract_offer_aid_effect = yes
            }
        }
    }
