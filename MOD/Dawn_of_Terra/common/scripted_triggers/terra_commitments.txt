# All event wrappers use ROOT/current and FROM as the actual negotiators.
# Weekly processing requires ROOT=THIS. No mutable country event targets exist.
terra_contract_free = {
    terra_dip_ready = yes
    terra_dip_sovereign = yes
    NOT = { has_country_flag = terra_contract_pending }
    NOT = { has_country_flag = terra_contract_active }
}
terra_contract_open_from = {
    terra_contract_free = yes
    FROM = { terra_contract_free = yes }
    terra_dip_talks_available_from = yes
    NOT = { terra_dip_conflicting_blocs_from = yes }
    NOT = { has_country_flag = terra_contract_offer_cooldown }
    FROM = { NOT = { has_country_flag = terra_contract_incoming_cooldown } }
}
terra_contract_pending_from = {
    terra_dip_talks_available_from = yes
    NOT = { terra_dip_conflicting_blocs_from = yes }
    has_country_flag = terra_contract_pending
    FROM = { has_country_flag = terra_contract_pending }
    terra_contract_has_role_with = { PARTNER = FROM ROLE = pending_with }
    FROM = { terra_contract_has_role_with = { PARTNER = ROOT ROLE = pending_with } }
}
terra_contract_aid_client_from = {
    terra_contract_pending_from = yes
    has_country_flag = terra_contract_pending_aid_client
    FROM = { has_country_flag = terra_contract_pending_aid_donor }
}
terra_contract_aid_donor_from = {
    terra_contract_pending_from = yes
    has_country_flag = terra_contract_pending_aid_donor
    FROM = { has_country_flag = terra_contract_pending_aid_client }
}
terra_contract_defense_client_from = {
    terra_contract_pending_from = yes
    has_country_flag = terra_contract_pending_protected
    FROM = { has_country_flag = terra_contract_pending_defender }
}
terra_contract_defense_donor_from = {
    terra_contract_pending_from = yes
    has_country_flag = terra_contract_pending_defender
    FROM = { has_country_flag = terra_contract_pending_protected }
    NOT = { has_guaranteed = FROM }
    DIPLOMACY_GUARANTEE_ENABLE_TRIGGER = yes
}
terra_contract_pair_valid = {
    terra_dip_ready = yes
    terra_dip_sovereign = yes
    $TARGET$ = { terra_dip_ready = yes terra_dip_sovereign = yes }
    NOT = { has_war_with = $TARGET$ }
    terra_contract_faction_unchanged = yes
    $TARGET$ = { terra_contract_faction_unchanged = yes }
    OR = {
        is_in_faction = no
        $TARGET$ = { is_in_faction = no }
        is_in_faction_with = $TARGET$
    }
}
terra_contract_aiding_defense_of = {
    any_country = {
        $TARGET$ = { has_defensive_war_with = PREV }
        ROOT = { has_war_with = PREV }
    }
}
terra_contract_defense_obligation_to = {
    terra_dip_ready = yes
    is_ai = yes
    has_country_flag = terra_contract_defender_$TARGET$
    has_guaranteed = $TARGET$
    terra_contract_pair_valid = { TARGET = $TARGET$ }
    $TARGET$ = { has_defensive_war = yes }
}
