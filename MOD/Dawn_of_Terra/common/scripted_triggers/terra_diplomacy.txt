terra_dip_sovereign = { terra_dip_participant = yes is_subject = no }
terra_dip_ready = { terra_dip_participant = yes has_country_flag = terra_dip_initialized }
terra_dip_exhausted = { check_variable = { terra_dip_pressure > 49 } }
terra_dip_military_policy_allowed = {
    terra_dip_ready = yes
    terra_ai_profile_allowed = yes
    is_subject = no
}

terra_dip_pair_valid = {
    terra_dip_ready = yes
    NOT = { tag = $TARGET$ }
    country_exists = $TARGET$
    $TARGET$ = { terra_dip_participant = yes }
}

terra_dip_conflicting_blocs_from = {
    is_in_faction = yes
    FROM = { is_in_faction = yes }
    NOT = { is_in_faction_with = FROM }
}

terra_dip_claim_conflict_from = {
    OR = {
        FROM = { any_owned_state = { is_core_of = ROOT } }
        any_owned_state = { is_core_of = FROM }
    }
}

terra_dip_has_common_interest_from = {
    OR = {
        is_in_faction_with = FROM
        has_war_together_with = FROM
        has_non_aggression_pact_with = FROM
        AND = {
            is_neighbor_of = FROM
            OR = {
                has_country_flag = terra_dip_personality_mercantile
                has_country_flag = terra_dip_personality_pragmatic
                FROM = { has_country_flag = terra_dip_personality_mercantile }
                FROM = { has_country_flag = terra_dip_personality_pragmatic }
            }
        }
    }
}

terra_dip_talks_available_from = {

    terra_dip_sovereign = yes
    FROM = { terra_dip_ready = yes terra_dip_sovereign = yes NOT = { tag = ROOT } }
    NOT = { has_war_with = FROM }

terra_dip_restraint_available_from = {

    NOT = { has_non_aggression_pact_with = FROM }
    NOT = { is_in_faction_with = FROM }
    NOT = { terra_dip_conflicting_blocs_from = yes }
}

terra_dip_can_prepare_against = {
    terra_dip_pair_valid = { TARGET = $TARGET$ }
    terra_dip_military_policy_allowed = yes
    has_war = no
    NOT = { terra_dip_exhausted = yes }
    terra_war_can_sustain_operations = yes
    terra_security_warning_ready = { TARGET = $TARGET$ }
    NOT = { has_country_flag = terra_state_goal_recovery }
    NOT = { is_in_faction_with = $TARGET$ }
    NOT = { has_non_aggression_pact_with = $TARGET$ }
    $TARGET$ = { is_subject = no }
    OR = {
        is_neighbor_of = $TARGET$
        $TARGET$ = { any_owned_state = { is_core_of = ROOT } }
    }
    OR = {
        check_variable = { terra_dip_grievance_$TARGET$ > 64 }
        AND = {
            has_country_flag = terra_dip_personality_imperial
            check_variable = { terra_dip_score_$TARGET$ < -54 }
        }
        AND = {
            $TARGET$ = { any_owned_state = { is_core_of = ROOT } }
            check_variable = { terra_dip_score_$TARGET$ < -24 }
        }
    }
}

# Hatred is a motive, not permission or military capability. Native war rules remain authoritative.
terra_dip_can_seek_war_against = {
    terra_dip_can_prepare_against = { TARGET = $TARGET$ }
    has_stability > 0.44
    surrender_progress < 0.05
    num_of_military_factories > 3
    num_divisions > 5
    has_manpower > 20000
    has_equipment = { infantry_equipment > 1500 }
    terra_security_confident_advantage = { TARGET = $TARGET$ }
    # A weak isolated victim must not hide a much stronger defending coalition.
    NOT = {
        $TARGET$ = {
            any_other_country = {
                OR = { is_in_faction_with = PREV has_guaranteed = PREV }
                NOT = { tag = ROOT }
                NOT = { strength_ratio = { tag = ROOT ratio < 0.8 } }
            }
        }
    }
}
