# All pair effects and helpers require ROOT = THIS leader, TARGET a concrete tag.
terra_coalition_actor = {
    terra_dip_ready = yes
    terra_ai_profile_allowed = yes
    is_ai = yes
    is_subject = no
    has_war = yes
    has_rule = can_join_factions
    NOT = { has_country_flag = terra_coalition_rejoin_cooldown }
}

terra_coalition_common_enemy = {
    any_country = {
        terra_dip_participant = yes
        NOT = { original_tag = TNJ }
        has_war_with = ROOT
        has_war_with = $TARGET$
    }
}

terra_coalition_pair_eligible = {
    terra_coalition_actor = yes
    terra_dip_pair_valid = { TARGET = $TARGET$ }
    $TARGET$ = { terra_coalition_actor = yes is_in_faction = no }
    NOT = { has_war_with = $TARGET$ }
    NOT = { has_wargoal_against = $TARGET$ }
    $TARGET$ = { NOT = { has_wargoal_against = ROOT } }
    has_opinion = { target = $TARGET$ value > -50 }
    $TARGET$ = { has_opinion = { target = ROOT value > -50 } }
    check_variable = { terra_dip_grievance_$TARGET$ < 70 }
    OR = {
        is_in_faction = no
        AND = { is_faction_leader = yes has_country_flag = terra_coalition_generated }
    }
    terra_coalition_common_enemy = { TARGET = $TARGET$ }
}
