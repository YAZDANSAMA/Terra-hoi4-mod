terra_ai_profile_allowed = {
    NOT = { OR = { original_tag = SST original_tag = TNJ original_tag = SEA } }
}
terra_ai_civilian_profile_allowed = {
    NOT = { OR = { original_tag = SST original_tag = SEA } }
}
terra_ai_active = {
    is_ai = yes has_capitulated = no any_owned_state = { always = yes }
}
terra_ai_can_expand_army = {
    terra_ai_active = yes num_of_military_factories > 0
    has_manpower > 20000 has_equipment = { infantry_equipment > 1500 }
}

# Training spends stored weapons, not factory slots. Aid-funded countries with
# no military industry must be able to train the same six-battalion foundation.
terra_ai_can_train_infantry = {
    terra_ai_profile_allowed = yes
    terra_ai_active = yes
    has_tech = infantry_weapons
    terra_ai_growth_reserves_ready = yes
}

terra_ai_growth_reserves_ready = {
    OR = {
        AND = {
            has_variable = terra_ai_manpower_reserve
            has_variable = terra_ai_rifle_reserve
            has_manpower > terra_ai_manpower_reserve
            has_equipment = { infantry_equipment > terra_ai_rifle_reserve }
        }
        AND = {
            NOT = { has_variable = terra_ai_manpower_reserve }
            has_manpower > 7500
            has_equipment = { infantry_equipment > 750 }
        }
    }
}

# A bounded country-sized training goal; this is an additional request, never
# a hard engine cap. The budget is refreshed at startup and weekly.
terra_ai_training_headroom = {
    OR = {
        AND = { has_variable = terra_ai_training_target check_variable = { num_divisions < terra_ai_training_target } }
        AND = { NOT = { has_variable = terra_ai_training_target } num_divisions < 24 }
    }
}

terra_ai_needs_train_line = {
    terra_ai_profile_allowed = yes
    terra_ai_active = yes
    has_tech = basic_train
    num_divisions > 5
    num_of_military_factories > 0
    OR = {
        AND = { num_of_military_factories > 3 has_equipment = { train_equipment < 10 } has_equipment = { infantry_equipment > 300 } }
        AND = { num_of_military_factories < 4 has_equipment = { train_equipment < 5 } has_equipment = { infantry_equipment > 1500 } }
    }
}

terra_ai_air_ready = {
    terra_ai_profile_allowed = yes
    terra_ai_active = yes
    has_war = yes
    air_base > 0
    fuel_ratio > 0.10
}

terra_ai_land_design_budget = { is_ai = yes num_of_military_factories > 5 has_equipment = { infantry_equipment > 1500 } }
terra_ai_air_design_budget = { is_ai = yes num_of_military_factories > 4 has_equipment = { infantry_equipment > 750 } }
terra_ai_naval_design_budget = { is_ai = yes num_of_naval_factories > 0 naval_base > 0 }
terra_ai_can_upgrade_infantry = {
    terra_ai_active = yes num_of_military_factories > 5 has_manpower > 20000
    has_tech = tech_engineers has_tech = interwar_artillery
    OR = {
        AND = { num_divisions < 11 has_equipment = { infantry_equipment > 3500 } has_equipment = { support_equipment > 250 } has_equipment = { artillery_equipment > 150 } }
        AND = { num_divisions > 10 num_divisions < 31 has_equipment = { infantry_equipment > 10000 } has_equipment = { support_equipment > 750 } has_equipment = { artillery_equipment > 450 } }
        AND = { num_divisions > 30 has_equipment = { infantry_equipment > 20000 } has_equipment = { support_equipment > 1500 } has_equipment = { artillery_equipment > 1000 } }
    }
}
