# Ordinary native AI pacts need earned confidence and a specific wartime use.
# Authored player restraint negotiations remain a separate explicit agreement.
terra_dip_can_offer_restraint = {
    terra_dip_pair_valid = { TARGET = $TARGET$ }
    is_subject = no
    $TARGET$ = { is_subject = no }
    is_neighbor_of = $TARGET$
    NOT = { has_war_with = $TARGET$ }
    NOT = { has_war_together_with = $TARGET$ }
    NOT = { is_in_faction_with = $TARGET$ }
    NOT = { has_non_aggression_pact_with = $TARGET$ }
    NOT = { has_wargoal_against = $TARGET$ }
    $TARGET$ = { NOT = { has_wargoal_against = ROOT } }
    NOT = { $TARGET$ = { any_owned_state = { is_core_of = ROOT } } }
    NOT = { any_owned_state = { is_core_of = $TARGET$ } }
    NOT = { has_country_flag = terra_security_warned_$TARGET$ }
    NOT = { has_country_flag = terra_security_threat_$TARGET$ }
    check_variable = { terra_dip_trust_$TARGET$ > 39 }
    check_variable = { terra_dip_score_$TARGET$ > 39 }
    check_variable = { terra_dip_grievance_$TARGET$ < 20 }
    has_opinion = { target = $TARGET$ value > 25 }
    $TARGET$ = { has_opinion = { target = ROOT value > 25 } }
    # Shared ideology, adjacency or a cautious ruler alone cannot qualify.
    OR = {
        has_war = yes
        check_variable = { terra_state_fatigue > 49 }
    }
}
