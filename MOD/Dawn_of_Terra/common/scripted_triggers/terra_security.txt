# Security supplements native diplomacy; it never grants permission to declare war.
# Motive/intelligence helpers require ROOT = THIS country. Cached shared-threat
# flags make the coalition helper safe inside another country's candidate scope.
terra_security_actor_allowed = {
    terra_dip_military_policy_allowed = yes
}

terra_security_has_escalation_motive = {
    terra_dip_pair_valid = { TARGET = $TARGET$ }
    OR = {
        is_neighbor_of = $TARGET$
        $TARGET$ = { any_owned_state = { is_core_of = ROOT } }
    }
    OR = {
        check_variable = { terra_dip_grievance_$TARGET$ > 64 }
        AND = {
            has_country_flag = terra_dip_personality_imperial
            check_variable = { terra_dip_score_$TARGET$ < -54 }
        }
        AND = {
            $TARGET$ = { any_owned_state = { is_core_of = ROOT } }
            check_variable = { terra_dip_score_$TARGET$ < -24 }
        }
    }
}

terra_security_can_warn = {
    is_ai = yes
    terra_security_actor_allowed = yes
    terra_security_has_escalation_motive = { TARGET = $TARGET$ }
    has_war = no
    NOT = { is_in_faction_with = $TARGET$ }
    NOT = { has_non_aggression_pact_with = $TARGET$ }
    NOT = { has_country_flag = terra_state_goal_recovery }
    check_variable = { terra_state_fatigue < 60 }
    check_variable = { terra_state_defeat_memory < 75 }
    check_variable = { terra_state_domestic_pressure < 70 }
    num_divisions > 3
    $TARGET$ = { is_subject = no NOT = { original_tag = TNJ } }
}

terra_security_warning_ready = {
    terra_security_can_warn = { TARGET = $TARGET$ }
    has_country_flag = terra_security_warned_$TARGET$
    has_country_flag = terra_security_warning_mature_$TARGET$
    NOT = { has_country_flag = terra_security_warning_pending_$TARGET$ }
}

# Native army strength is an estimate. Lower observer intelligence requires a
# larger margin, and even high confidence is never described as certain.
terra_security_confident_advantage = {
    terra_dip_pair_valid = { TARGET = $TARGET$ }
    OR = {
        AND = {
            has_country_flag = terra_security_intel_high_$TARGET$
            strength_ratio = { tag = $TARGET$ ratio > 1.35 }
        }
        AND = {
            has_country_flag = terra_security_intel_medium_$TARGET$
            strength_ratio = { tag = $TARGET$ ratio > 1.5 }
        }
        AND = {
            NOT = { has_country_flag = terra_security_intel_high_$TARGET$ }
            NOT = { has_country_flag = terra_security_intel_medium_$TARGET$ }
            strength_ratio = { tag = $TARGET$ ratio > 1.8 }
        }
    }
}

terra_security_public_danger = {
    terra_dip_ready = yes
    OR = {
        has_country_flag = terra_dip_recent_aggressor
        has_country_flag = terra_dip_oathbreaker
        AND = {
            check_variable = { terra_state_reputation < 35 }
            OR = { has_offensive_war = yes has_country_flag = terra_state_goal_revision }
        }
    }
}

terra_security_assesses_threat = {
    terra_security_actor_allowed = yes
    terra_dip_pair_valid = { TARGET = $TARGET$ }
    NOT = { is_in_faction_with = $TARGET$ }
    NOT = { has_non_aggression_pact_with = $TARGET$ }
    $TARGET$ = { terra_security_public_danger = yes NOT = { original_tag = TNJ } }
    OR = {
        is_neighbor_of = $TARGET$
        $TARGET$ = { any_owned_state = { is_core_of = ROOT } }
    }
    OR = {
        has_war_with = $TARGET$
        check_variable = { terra_dip_score_$TARGET$ < -24 }
        check_variable = { terra_dip_grievance_$TARGET$ > 44 }
        $TARGET$ = { has_wargoal_against = ROOT }
        $TARGET$ = { any_owned_state = { is_core_of = ROOT } }
    }
}
