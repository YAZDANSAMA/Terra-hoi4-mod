on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = {
					country_exists = LGL
					country_exists = VCT
				}
				VCT = {
					add_opinion_modifier = {
						target = LGL
						modifier = VCT_LGL_reps
					}
				}
                LGL = {
					add_opinion_modifier = {
						target = VCT
						modifier = LGL_VCT_reps
					}
				}
			}
		}
	}

	on_daily = {
		effect = {
			if = {
				limit = { tag = CLM }
				if = {
					limit = {
						check_variable = { CLM_run_total > 0 }
						OR = {
							has_country_flag = CLM_electoral_busy
							has_active_mission = CLM_fund_underdeveloped_mission
							has_active_mission = CLM_infra_grants_mission
						}
					}
					add_to_variable = { CLM_run_days_left = -1 }
					clamp_variable = { var = CLM_run_days_left min = 0 max = 180 }
					set_variable = { CLM_run_frame = CLM_run_total }
					subtract_from_variable = { CLM_run_frame = CLM_run_days_left }
					multiply_variable = { CLM_run_frame = 15 }
					divide_variable = { CLM_run_frame = CLM_run_total }
					round_variable = CLM_run_frame
					add_to_variable = { CLM_run_frame = 1 }
				}
			}
		}
	}
}
