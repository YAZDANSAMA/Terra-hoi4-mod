# Signed defensive commitments override ordinary optional-call caution.
# Preference applies to AI only; native legality remains authoritative.

terra_contract_honour_defense_AGI = {
    allowed = { NOT = { OR = { original_tag = SST original_tag = SEA original_tag = TNJ } } }
    enable = {
        is_ai = yes terra_dip_ready = yes
        terra_dip_pair_valid = { TARGET = AGI }
        terra_contract_defense_obligation_to = { TARGET = AGI }
        AGI = { has_defensive_war = yes }
        NOT = { has_war_together_with = AGI }
    }
    abort_when_not_enabled = yes
    ai_strategy = { type = diplo_action_acceptance id = AGI target = call_allies value = 160 }
    ai_strategy = { type = diplo_action_desire id = AGI target = join_allies value = 160 }
}
