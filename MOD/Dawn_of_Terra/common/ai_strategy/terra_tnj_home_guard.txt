# Ta-Netjer's army is a permanent sanctuary guard, not an expeditionary force.
# Native template/country locks enforce recruitment restrictions. The fronts
# and buffers below control AI orders; they are not a geographic engine lock.

terra_tnj_home_guard = {
    allowed = { original_tag = TNJ }
    enable = { is_ai = yes has_capitulated = no num_of_owned_states > 0 }
    abort_when_not_enabled = yes

    # All ten remain in the four home states, even during an allied war.
    # Do not release this reserve to service foreign fronts or invasions.
    ai_strategy = {
        type = put_unit_buffers
        ratio = 1.0
        states = { 170 172 179 180 }
        subtract_invasions_from_need = no
        subtract_fronts_from_need = no
    }
    ai_strategy = { type = garrison id = TNJ value = 100 }
    ai_strategy = { type = garrison_reinforcement_priority value = 100 }
    ai_strategy = { type = ai_wanted_divisions_factor value = -100 }

    # No offensive battleplan execution, manual attacks or naval landings.
    # Defending units still fight attackers normally.
    ai_strategy = {
        type = front_control
        country_trigger = { always = yes }
        ratio = 0.0
        priority = 1000
        ordertype = front
        execution_type = careful
        execute_order = no
        manual_attack = no
    }
    ai_strategy = {
        type = front_control
        country_trigger = { always = yes }
        ratio = 0.0
        priority = 1000
        ordertype = invasion
        execute_order = no
    }
    ai_strategy = {
        type = front_unit_request
        state_trigger = { NOT = { is_owned_by = TNJ } }
        value = -100
    }
    ai_strategy = {
        type = invasion_unit_request
        country_trigger = { always = yes }
        value = -100
    }
}

# Unlike divisions, surplus equipment may support the empire. Native stockpile,
# diplomatic, war and transport checks still decide whether aid is sent.
terra_tnj_equipment_aid_SGN = {
    allowed = { original_tag = TNJ }
    enable = {
        is_ai = yes has_capitulated = no has_war = no
        country_exists = SGN SGN = { has_war = yes has_capitulated = no }
        OR = { is_subject_of = SGN is_in_faction_with = SGN }
        has_equipment = { infantry_equipment > 1500 }
    }
    abort_when_not_enabled = yes
    ai_strategy = { type = send_lend_lease_desire id = SGN value = 25 }
}
