# Suppress generic AI pact offers and acceptance outside earned, useful restraint.

terra_dip_reject_generic_pact_AGI = {
    allowed = { NOT = { OR = { original_tag = SST original_tag = SEA } } }
    enable = {
        is_ai = yes terra_dip_pair_valid = { TARGET = AGI }
        NOT = { has_non_aggression_pact_with = AGI }
        NOT = { terra_dip_can_offer_restraint = { TARGET = AGI } }
    }
    abort_when_not_enabled = yes
    ai_strategy = { type = diplo_action_desire id = AGI target = non_aggression_pact value = -1000 }
    ai_strategy = { type = diplo_action_acceptance id = AGI target = non_aggression_pact value = -1000 }
}
