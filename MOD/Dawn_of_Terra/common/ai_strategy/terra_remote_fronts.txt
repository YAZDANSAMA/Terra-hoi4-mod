# Remote fronts require both access and a unit allocation request.
# Enemy country is country_trigger's scope; FROM is the AI country.
# Home defense wins when an uncapitulated enemy directly borders the AI.
terra_safe_remote_fronts = {
    allowed = { NOT = { OR = { original_tag = SST original_tag = SEA original_tag = TNJ } } }
    enable = {
        OR = { NOT = { terra_dip_ready = yes } terra_war_can_spare_allied_force = yes }
        is_ai = yes has_capitulated = no has_war = yes
        num_divisions > 2 surrender_progress < 0.10
        NOT = { any_enemy_country = { is_neighbor_of = ROOT has_capitulated = no } }
        any_country = {
            NOT = { tag = ROOT }
            has_war_together_with = ROOT
            NOT = { has_war_with = ROOT }
            OR = {
                is_subject_of = ROOT
                ROOT = { is_subject_of = PREV }
                is_in_faction_with = ROOT
                AND = { is_subject = yes overlord = { has_subject = ROOT } }
            }
        }
    }
    abort_when_not_enabled = yes
    ai_strategy = { type = spare_unit_factor value = 100 }
    ai_strategy = {
        type = front_unit_request
        country_trigger = { has_war_with = FROM NOT = { is_neighbor_of = FROM } }
        value = 100
    }
}
