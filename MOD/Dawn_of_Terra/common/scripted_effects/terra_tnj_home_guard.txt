# AI orders alone cannot enforce a geographic boundary. This country-scoped
# callback also repairs old saves and returns only TNJ-owned armies abroad.
terra_tnj_enforce_home_guard_effect = {
    if = {
        limit = {
            tag = TNJ original_tag = TNJ is_ai = yes exists = yes
            has_capitulated = no
        }
        if = {
            limit = { NOT = { has_country_flag = terra_tnj_home_policy_migrated } }
            set_rule = {
                desc = TNJ_home_guard_policy
                can_be_called_to_war = no
                can_decline_call_to_war = yes
                can_send_volunteers = no
                can_not_declare_war = yes
                units_deployed_to_overlord = no
            }
            country_lock_all_division_template = { is_locked = yes desc = TNJ_home_guard_limit }
            set_country_flag = terra_tnj_home_policy_migrated
        }
        # No safe destination means no teleport: do not rescue an occupied or
        # capitulated country by moving units into enemy-controlled territory.
        random_owned_controlled_state = {
            every_state = {
                limit = { NOT = { is_owned_by = TNJ } }
                teleport_armies = {
                    limit = { tag = TNJ }
                    to_state = PREV
                }
            }
        }
    }
}
