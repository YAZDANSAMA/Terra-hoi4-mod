# Only eligible AI governments already fighting a common enemy are enrolled.
# ROOT=THIS; TARGET is always an explicit registry tag. No player or existing bloc is moved.
terra_coalition_create_named_effect = {
    if = {
        limit = { terra_coalition_actor = yes is_in_faction = no }
        # Emergency wartime formation grants permission only to the chosen leader.
        set_rule = { can_create_factions = yes }
        if = {
            limit = { has_dlc = "No Compromise, No Surrender" }
            create_faction_from_template = { template = terra_faction_template_common_defense name = $NAME$ }
        }
        else = { create_faction = $NAME$ }
        if = { limit = { is_faction_leader = yes } set_country_flag = terra_coalition_generated }
    }
}

terra_coalition_form_pair_effect = {
    if = {
        limit = { terra_coalition_pair_eligible = { TARGET = $TARGET$ } }
        if = { limit = { is_in_faction = no } terra_coalition_choose_name_effect = yes }
        # Recheck after creation, before introducing the member. A failed create is harmless.
        if = {
            limit = {
                is_faction_leader = yes has_country_flag = terra_coalition_generated
                terra_coalition_pair_eligible = { TARGET = $TARGET$ }
            }
            add_to_faction = $TARGET$
            set_country_flag = terra_coalition_member
            $TARGET$ = { set_country_flag = terra_coalition_member }
            terra_dip_change_relation_effect = { PARTNER = $TARGET$ TRUST = 10 GRIEVANCE = -2 }
            $TARGET$ = { terra_dip_change_relation_effect = { PARTNER = ROOT TRUST = 10 GRIEVANCE = -2 } }
        }
    }
}
