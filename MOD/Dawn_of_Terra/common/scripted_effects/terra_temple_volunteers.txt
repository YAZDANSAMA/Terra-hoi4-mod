# Off-map historical reservists supplement normal core-population recruitment.
# They do not change state population, laws or periodic recruitment.
# Country history calls these only at campaign setup; saved flags make an
# accidental second call harmless. Old saves are never topped up on startup.
terra_sgn_initialize_reserves_effect = {
    if = {
        limit = {
            tag = SGN
            NOT = { has_country_flag = terra_sgn_starting_reserves_set }
        }
        # 27,525,293 * 0.025 + 1,479,868 - 168,000 in the army
        # leaves approximately 2,000,000 free recruits at the 1028 start.
        add_manpower = 1479868
        set_country_flag = terra_sgn_starting_reserves_set
    }
}

    if = {
        limit = { terra_tnj_can_send_volunteers = yes }
        set_variable = { terra_tnj_last_transfer_day = num_days }
        add_manpower = -2000
