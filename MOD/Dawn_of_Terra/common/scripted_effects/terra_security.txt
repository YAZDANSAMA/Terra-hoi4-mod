# ROOT = THIS for weekly/monthly/motive/intelligence effects.
# Warning cooldown survives a backdown; escalation notice does not.
terra_security_refresh_intel_pair_effect = {
    clr_country_flag = terra_security_intel_low_$TARGET$
    clr_country_flag = terra_security_intel_medium_$TARGET$
    clr_country_flag = terra_security_intel_high_$TARGET$
    set_variable = { terra_security_intel_$TARGET$ = 1 }
    if = {
        limit = { terra_dip_pair_valid = { TARGET = $TARGET$ } }
        if = {
            limit = {
                OR = {
                    AND = {
                        has_dlc = "La Resistance"
                        intel_level_over = { target = $TARGET$ army_intel > 0.65 }
                    }
                    AND = {
                        NOT = { has_dlc = "La Resistance" }
                        has_tech = advanced_decryption
                        $TARGET$ = { NOT = { has_tech = advanced_encryption } }
                    }
                }
            }
            set_country_flag = terra_security_intel_high_$TARGET$
            set_variable = { terra_security_intel_$TARGET$ = 3 }
        }
        else_if = {
            limit = {
                OR = {
                    AND = {
                        has_dlc = "La Resistance"
                        intel_level_over = { target = $TARGET$ army_intel > 0.35 }
                    }
                    AND = {
                        NOT = { has_dlc = "La Resistance" }
                        OR = { has_tech = basic_decryption has_tech = improved_decryption has_tech = advanced_decryption }
                        $TARGET$ = { NOT = { has_tech = advanced_encryption } }
                    }
                }
            }
            set_country_flag = terra_security_intel_medium_$TARGET$
            set_variable = { terra_security_intel_$TARGET$ = 2 }
        }
        else = { set_country_flag = terra_security_intel_low_$TARGET$ }
    }
    else = { set_country_flag = terra_security_intel_low_$TARGET$ }
}

terra_security_refresh_warning_pair_effect = {
    if = {
        limit = { terra_security_can_warn = { TARGET = $TARGET$ } }
        if = {
            limit = {
                NOT = { has_country_flag = terra_security_warned_$TARGET$ }
                NOT = { has_country_flag = terra_security_warning_cooldown_$TARGET$ }
                OR = {
                    $TARGET$ = { is_ai = yes }
                    $TARGET$ = { is_ai = no NOT = { has_country_flag = terra_security_notice_cooldown } }
                }
            }
            set_country_flag = terra_security_warned_$TARGET$
            clr_country_flag = terra_security_warning_mature_$TARGET$
            set_country_flag = { flag = terra_security_warning_pending_$TARGET$ value = 1 days = 60 }
            set_country_flag = { flag = terra_security_warning_cooldown_$TARGET$ value = 1 days = 180 }
            if = {
                limit = { $TARGET$ = { is_ai = no } }
                $TARGET$ = {
                    set_country_flag = { flag = terra_security_notice_cooldown value = 1 days = 30 }
                    country_event = { id = terra_security.1 }
                }
            }
        }
        if = {
            limit = {
                has_country_flag = terra_security_warned_$TARGET$
                NOT = { has_country_flag = terra_security_warning_pending_$TARGET$ }
            }
            set_country_flag = terra_security_warning_mature_$TARGET$
        }
    }
    else = {
        clr_country_flag = terra_security_warned_$TARGET$
        clr_country_flag = terra_security_warning_pending_$TARGET$
        clr_country_flag = terra_security_warning_mature_$TARGET$
    }
}

terra_security_refresh_threat_pair_effect = {

    else = { clr_country_flag = terra_security_threat_$TARGET$ }
}

        set_country_flag = terra_security_coalition_$TARGET$
    }
    else = { clr_country_flag = terra_security_coalition_$TARGET$ }
}

terra_security_monthly_effect = {
    if = {
        limit = {
            is_ai = yes
            terra_security_actor_allowed = yes
            NOT = { has_country_flag = terra_security_outreach_cooldown }
            NOT = { has_country_flag = terra_state_goal_revision }
            check_variable = { terra_state_fatigue < 70 }
        }
        random_country = {
            limit = {
                is_ai = no
                terra_security_actor_allowed = yes
                NOT = { has_country_flag = terra_security_invitation_cooldown }
                terra_security_can_coordinate = { PARTNER = ROOT }
            }
            set_country_flag = { flag = terra_security_invitation_cooldown value = 1 days = 120 }
            ROOT = { set_country_flag = { flag = terra_security_outreach_cooldown value = 1 days = 180 } }
            country_event = { id = terra_security.10 }
        }
    }
}
