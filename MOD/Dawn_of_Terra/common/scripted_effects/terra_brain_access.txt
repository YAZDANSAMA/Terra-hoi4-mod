# Current scope is the host. At least one participant must be AI.
# Pair-specific flags mark only permissions created by this feature.
terra_brain_access_reconcile_pair_effect = {
    if = {
        limit = { terra_brain_access_pair_eligible = { GUEST = $GUEST$ } }
        if = {
            limit = {
                NOT = { gives_military_access_to = $GUEST$ }
            }
            give_military_access = $GUEST$
            # Own a route only after the native access trigger confirms it.
            if = {
                limit = { gives_military_access_to = $GUEST$ }
                set_country_flag = terra_brain_access_granted_$GUEST$
            }
        }
        if = {
            limit = { has_country_flag = terra_brain_access_granted_$GUEST$ }
            set_country_flag = terra_brain_access_has_routes
        }
    }
    else_if = {
        limit = { has_country_flag = terra_brain_access_granted_$GUEST$ }
        if = {
            limit = {
                $GUEST$ = { exists = yes }
                gives_military_access_to = $GUEST$
            }
            # military_access diplomatic_relation is removed in the guest's
            # scope: cancel that country's permission to enter this host.
            # Do not revoke the host's independent access in the other direction.
            $GUEST$ = {
                diplomatic_relation = {
                    country = PREV
                    relation = military_access
                    active = no
                }
            }
        }
        clr_country_flag = terra_brain_access_granted_$GUEST$
    }
}
