terra_dip_refresh_pressure_effect = {
    set_variable = { terra_dip_pressure = 0 }
    if = { limit = { has_war = yes } add_to_variable = { terra_dip_pressure = 20 } }
    if = { limit = { has_stability < 0.4 } add_to_variable = { terra_dip_pressure = 35 } }
    if = { limit = { surrender_progress > 0.15 } add_to_variable = { terra_dip_pressure = 35 } }
    if = { limit = { num_divisions > 3 has_manpower < 10000 } add_to_variable = { terra_dip_pressure = 25 } }
    if = { limit = { num_divisions > 3 has_equipment = { infantry_equipment < 200 } } add_to_variable = { terra_dip_pressure = 20 } }
    if = { limit = { check_variable = { terra_state_fatigue > 49 } } add_to_variable = { terra_dip_pressure = 25 } }
    if = { limit = { check_variable = { terra_state_defeat_memory > 59 } } add_to_variable = { terra_dip_pressure = 20 } }

terra_dip_refresh_pair_effect = {
    if = {
        limit = { terra_dip_pair_valid = { TARGET = $TARGET$ } }
        # Detect actual abandonment, not an ordinary peaceful change of diplomatic alignment.
        if = {
            limit = {
                has_country_flag = terra_dip_partner_$TARGET$
                NOT = { is_in_faction_with = $TARGET$ }
            }
            if = {
                limit = { has_war = yes NOT = { has_war_together_with = $TARGET$ } }
                terra_dip_change_pair_effect = { TARGET = $TARGET$ TRUST = -35 GRIEVANCE = 30 }
                add_opinion_modifier = { target = $TARGET$ modifier = terra_dip_betrayal days = 720 }
            }
            else = {
                terra_dip_change_pair_effect = { TARGET = $TARGET$ TRUST = -15 GRIEVANCE = 10 }
                add_opinion_modifier = { target = $TARGET$ modifier = terra_dip_broken_compact days = 180 }
            }
            clr_country_flag = terra_dip_partner_$TARGET$
        }
        if = {
            limit = { is_in_faction_with = $TARGET$ }
            set_country_flag = terra_dip_partner_$TARGET$
        }
        if = {
            limit = { has_non_aggression_pact_with = $TARGET$ }
            set_country_flag = terra_dip_pact_$TARGET$
        }
        else = { clr_country_flag = terra_dip_pact_$TARGET$ }

        set_variable = { terra_dip_score_$TARGET$ = terra_dip_trust_$TARGET$ }
        subtract_from_variable = { terra_dip_score_$TARGET$ = terra_dip_grievance_$TARGET$ }
        if = { limit = { is_in_faction_with = $TARGET$ } add_to_variable = { terra_dip_score_$TARGET$ = 35 } }
        if = {
            limit = { OR = { is_subject_of = $TARGET$ $TARGET$ = { is_subject_of = ROOT } } }
            add_to_variable = { terra_dip_score_$TARGET$ = 30 }
        }
        if = { limit = { has_war_together_with = $TARGET$ } add_to_variable = { terra_dip_score_$TARGET$ = 20 } }
        if = { limit = { has_non_aggression_pact_with = $TARGET$ } add_to_variable = { terra_dip_score_$TARGET$ = 10 } }
        if = { limit = { has_government = $TARGET$ } add_to_variable = { terra_dip_score_$TARGET$ = 10 } }
        if = { limit = { has_opinion = { target = $TARGET$ value > 25 } } add_to_variable = { terra_dip_score_$TARGET$ = 15 } }
        if = { limit = { has_opinion = { target = $TARGET$ value > 75 } } add_to_variable = { terra_dip_score_$TARGET$ = 15 } }
        if = { limit = { has_opinion = { target = $TARGET$ value < -25 } } add_to_variable = { terra_dip_score_$TARGET$ = -25 } }
        if = { limit = { has_opinion = { target = $TARGET$ value < -75 } } add_to_variable = { terra_dip_score_$TARGET$ = -25 } }
        if = {
            limit = {
                is_neighbor_of = $TARGET$
                OR = { has_country_flag = terra_dip_personality_mercantile has_country_flag = terra_dip_personality_pragmatic }
                NOT = { has_war_with = $TARGET$ }
            }
            add_to_variable = { terra_dip_score_$TARGET$ = 12 }
        }
        if = {
            limit = { $TARGET$ = { any_owned_state = { is_core_of = ROOT } } }
            add_to_variable = { terra_dip_score_$TARGET$ = -30 }
        }
        if = {
            limit = { is_in_faction = yes $TARGET$ = { is_in_faction = yes } NOT = { is_in_faction_with = $TARGET$ } }
            add_to_variable = { terra_dip_score_$TARGET$ = -15 }
        }
        if = {
            limit = { $TARGET$ = { has_country_flag = terra_dip_recent_aggressor } NOT = { is_in_faction_with = $TARGET$ } }
            add_to_variable = { terra_dip_score_$TARGET$ = -10 }
        }
        if = { limit = { $TARGET$ = { has_country_flag = terra_dip_oathbreaker } } add_to_variable = { terra_dip_score_$TARGET$ = -20 } }
        if = {
            limit = { $TARGET$ = { has_country_flag = terra_state_initialized check_variable = { terra_state_reputation < 30 } } }
            add_to_variable = { terra_dip_score_$TARGET$ = -15 }
        }
        if = {
            limit = { $TARGET$ = { has_country_flag = terra_state_initialized check_variable = { terra_state_reputation > 74 } } }
            add_to_variable = { terra_dip_score_$TARGET$ = 10 }
        }
        if = {
            limit = { has_country_flag = terra_state_goal_security is_in_faction_with = $TARGET$ }
            add_to_variable = { terra_dip_score_$TARGET$ = 10 }
        }
        if = {
            limit = { has_country_flag = terra_state_goal_commerce is_neighbor_of = $TARGET$ NOT = { has_war_with = $TARGET$ } }
            add_to_variable = { terra_dip_score_$TARGET$ = 8 }
        }
        if = { limit = { has_war_with = $TARGET$ } add_to_variable = { terra_dip_score_$TARGET$ = -50 } }
        clamp_variable = { var = terra_dip_score_$TARGET$ min = -100 max = 100 }
    }
    else = {
        set_variable = { terra_dip_score_$TARGET$ = 0 }
        # Annexation/absence must not be mistaken for a voluntary betrayal on later reappearance.
        clr_country_flag = terra_dip_partner_$TARGET$
        clr_country_flag = terra_dip_pact_$TARGET$
    }
}

        }
    }
}
