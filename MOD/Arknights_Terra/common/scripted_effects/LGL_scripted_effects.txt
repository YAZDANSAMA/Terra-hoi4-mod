LGL_recalc_military_spirit = {
    
    set_variable = { LGL_attack_var = 0 }
    set_variable = { LGL_defense_var = 0 }
    set_variable = { LGL_speed_var = 0 }
    set_variable = { LGL_training_var = 0.06 }
    set_variable = { LGL_recruitpop_var = 0 }
    set_variable = { LGL_org_var = -0.05 }
    set_variable = { LGL_xp_var = -0.1 }
    set_variable = { LGL_winter_var = 0 }
    set_variable = { LGL_planspd_var = -0.15 }
    set_variable = { LGL_maxplan_var = -0.1 }
    set_variable = { LGL_orgloss_var = 0 }
    set_variable = { LGL_attrition_var = 0 }
    set_variable = { LGL_supplygrace_var = 0 }

    # --- Asymmetric branch ---
    if = {
        limit = { has_country_flag = LGL_asym_1_done }
        add_to_variable = { LGL_orgloss_var = -0.025 }
        add_to_variable = { LGL_speed_var = 0.03 }
    }
    if = {
        limit = { has_country_flag = LGL_asym_2_done }
        add_to_variable = { LGL_attack_var = 0.05 }
        add_to_variable = { LGL_attrition_var = -0.15 }
        add_to_variable = { LGL_orgloss_var = -0.025 }
        add_to_variable = { LGL_supplygrace_var = 30 }

    }
    if = {
        limit = { has_country_flag = LGL_asym_3_done }
        add_to_variable = { LGL_attack_var = 0.10 }
        add_to_variable = { LGL_org_var = -0.10 }
    }

    # --- Static branch ---
    if = {
        limit = { has_country_flag = LGL_stat_1_done }
        add_to_variable = { LGL_org_var = 0.03 }
        add_to_variable = { LGL_training_var = -0.05 }
        add_to_variable = { LGL_xp_var = 0.10 }
    }
    if = {
        limit = { has_country_flag = LGL_stat_2_done }
        add_to_variable = { LGL_org_var = 0.05 }
        add_to_variable = { LGL_defense_var = 0.05 }
    }
    if = {
        limit = { has_country_flag = LGL_stat_3_done }
        add_to_variable = { LGL_speed_var = 0.075 }
        add_to_variable = { LGL_winter_var = -0.10 }
        add_to_variable = { LGL_planspd_var = 0.10 } 
    }


    # --- Shared branch ---
    if = {
        limit = { has_country_flag = LGL_shared_1_done }
        add_to_variable = { LGL_planspd_var = 0.15 } 
        add_to_variable = { LGL_maxplan_var = 0.125 }
        add_to_variable = { LGL_orgloss_var = -0.075 }
    }
    if = {
        limit = { has_country_flag = LGL_shared_2_done }
        add_to_variable = { LGL_training_var = -0.10 }
        add_to_variable = { LGL_recruitpop_var = 0.10 }
    }

    # --- Sarkaz Arts ---
    if = {
        limit = { has_country_flag = LGL_arts_1_done }
        add_to_variable = { LGL_attack_var = 0.08 }
        add_to_variable = { LGL_org_var = -0.03 }
    }
    if = {
        limit = { has_country_flag = LGL_arts_2_done }
        add_to_variable = { LGL_org_var = -0.05 }
        add_to_variable = { LGL_attack_var = 0.07 }
    }

    # --- Steam Knights ---
    if = {
        limit = { has_country_flag = LGL_steam_1_done }
        add_to_variable = { LGL_defense_var = 0.08 }
        add_to_variable = { LGL_attack_var = 0.04 }
    }
    if = {
        limit = { has_country_flag = LGL_steam_2_done }
        add_to_variable = { LGL_defense_var = 0.09 }
        add_to_variable = { LGL_org_var = 0.02 }
    }

    # --- Final ---
    if = {
        limit = { has_country_flag = LGL_mil_final_done }
        add_to_variable = { LGL_attack_var = 0.08 }
        add_to_variable = { LGL_org_var = 0.07 }
        add_to_variable = { LGL_recruitpop_var = 0.10 }
        add_to_variable = { LGL_xp_var = 0.10 }
    }

    force_update_dynamic_modifier = yes
}
